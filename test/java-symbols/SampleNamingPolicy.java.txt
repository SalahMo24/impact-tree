package demo;

import java.lang.reflect.Field;
import java.util.Locale;

/**
 * Ways to turn a field name into the name used in text.
 */
public enum NamingPolicy implements NamingStrategy {

  /** The field name unchanged. */
  IDENTITY() {
    @Override
    public String translateName(Field f) {
      return f.getName();
    }
  },

  /** The first letter in upper case. */
  UPPER_CAMEL_CASE() {
    @Override
    public String translateName(Field f) {
      return upperCaseFirst(f.getName());
    }
  },

  /** Upper camel case with a space between the words. */
  UPPER_CAMEL_CASE_WITH_SPACES() {
    @Override
    public String translateName(Field f) {
      return upperCaseFirst(separateWords(f.getName(), ' '));
    }
  },

  /** Upper case with an underscore between the words. */
  UPPER_CASE_WITH_UNDERSCORES() {
    @Override
    public String translateName(Field f) {
      return separateWords(f.getName(), '_').toUpperCase(Locale.ENGLISH);
    }
  },

  /** Lower case with an underscore between the words. */
  LOWER_CASE_WITH_UNDERSCORES() {
    @Override
    public String translateName(Field f) {
      return separateWords(f.getName(), '_').toLowerCase(Locale.ENGLISH);
    }
  },

  /** Lower case with a dash between the words. */
  LOWER_CASE_WITH_DASHES() {
    @Override
    public String translateName(Field f) {
      return separateWords(f.getName(), '-').toLowerCase(Locale.ENGLISH);
    }
  },

  /** Lower case with a dot between the words. */
  LOWER_CASE_WITH_DOTS() {
    @Override
    public String translateName(Field f) {
      return separateWords(f.getName(), '.').toLowerCase(Locale.ENGLISH);
    }
  };

  /**
   * Inserts the separator before each upper case letter, so that a camel case
   * name splits into separate words.
   */
  static String separateWords(String name, char separator) {
    StringBuilder translation = new StringBuilder();
    for (int i = 0, length = name.length(); i < length; i++) {
      char character = name.charAt(i);
      if (Character.isUpperCase(character) && translation.length() != 0) {
        translation.append(separator);
      }
      translation.append(character);
    }
    return translation.toString();
  }

  static String upperCaseFirst(String s) {
    if (s.isEmpty() || Character.isUpperCase(s.charAt(0))) {
      return s;
    }
    return Character.toUpperCase(s.charAt(0)) + s.substring(1);
  }
}
